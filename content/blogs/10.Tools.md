1.langchain内置工具(包)

在 LangChain 中，所有工具都是 `BaseTool` 的子类，并且工具也是 `Runnable` 可运行组件，支持通过 `invoke`、`batch` 和 `stream` 等方法进行调用，使用起来也非常简单，实例化工具后直接调用 `invoke` 函数即可，除此之外，可以通过 `name`、`description`、`args`、`return_direct` 属性来获取到工具的相关信息。

内置工具tools与内置工具包toolkits里面有很多现成工具。

## 1.1.内置工具DuckDuckGoSearchResults

```python
from langchain_community.tools import DuckDuckGoSearchRun,DuckDuckGoSearchResults
from langchain_core.utils.function_calling import convert_to_openai_tool

# pip install duckduckgo-search
# pip install -U ddgs

# 1.工具实例化
# search = DuckDuckGoSearchRun()
search = DuckDuckGoSearchResults()  # result比run返回的参数更多，封装更完整
print(search.invoke("今天的美国新闻"))  # invok e调用工具
print(search.run("今天的美国新闻"))  # run调用工具
print(convert_to_openai_tool(search))  # convert_to_openai_tool转换成openai等模型使用的tool的格式
print(search)
print("名字：", search.name)
print("描述：", search.description)
print("参数：", search.args)
print("是否直接返回：", search.return_direct)
```

# 2.自定义工具

在LangChain中提供了3种构建自定义工具的方法：@tool装饰器，StructuredTool.from_function()类方法，BaseTool子类。

## 2.1@tool装饰器

该方法必须提供文档字符串。

```python
from langchain_core.tools import tool
from pydantic import Field


class MultiplyInputs(BaseModel):
    a: int = Field(description="第一个数字")
    b: int = Field(description="第二个数字")


@tool("multiply_tool", return_direct=True, args_schema=MultiplyInputs)
def multiply(a: int, b: int) -> int:
    """将传递的两个数字相乘"""
    return a * b


print("名字：", multiply.name)
print("描述：", multiply.description)
print("参数：", multiply.args)
print("是否直接返回：", multiply.return_direct)
print(multiply.invoke({"a": 1, "b": 2}))
```

## 2.2.StructuredTool类方法

支持同步与异步方法。

```python
from langchain_core.tools import StructuredTool


class MultiplyInputs(BaseModel):
    a: int = Field(description="第一个数字")
    b: int = Field(description="第二个数字")


def multiply(a: int, b: int) -> int:
    """将传递的两个数字相乘"""
    return a * b


async def amultiply(a: int, b: int) -> int:
    """将传递的两个数字相乘"""
    return a * b


calculator = StructuredTool.from_function(
    func=multiply,
    coroutine=amultiply,
    name="multiply_tool",
    description="将传递的两个数字相乘",
    return_direct=True,
    args_schema=MultiplyInputs,
)
print("名字：", calculator.name)
print("描述：", calculator.description)
print("参数：", calculator.args)
print("是否直接返回：", calculator.return_direct)
print(calculator.invoke({"a": 1, "b": 2}))
```

## 2.3.BaseTool子类

在LangChain中所有的工具都BaseTool子类，所以以上两种方式都是包装成StruceuredTool，也就是BaseTool的子类。

```python
from typing import Type, Any
from langchain_core.tools import BaseTool


class MultiplyInputs(BaseModel):
    a: int = Field(description="第一个数字")
    b: int = Field(description="第二个数字")


class MultiplyTool(BaseTool):
    """乘法计算工具"""
    name: str = "multiply_tool"
    description: str = "两数相乘"
    args_schema: Type[BaseModel] = MultiplyInputs

    def _run(self, a: int, b: int) -> Any:
        return a * b


calculator = MultiplyTool()
print("名字：", calculator.name)
print("描述：", calculator.description)
print("参数：", calculator.args)
print("是否直接返回：", calculator.return_direct)
print(calculator.invoke({"a": 1, "b": 2}))
```

# 3.工具应用bind_tools

bind函数的二次封装。

```python
import os
from typing import Type, Any

import dotenv
from langchain_community.tools import GoogleSerperRun
from langchain_community.utilities import GoogleSerperAPIWrapper
from langchain_core.messages import ToolMessage
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough
from langchain_core.tools import BaseTool
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field


# 定义工具参数 Schema
class GoogleSerperArgsSchema(BaseModel):
    query: str = Field(description="执行谷歌搜索的查询语句")


# 加载环境变量（如 API Key）
dotenv.load_dotenv()

# 创建 Google Serper 搜索工具
google_serper = GoogleSerperRun(
    name="google_serper",
    description=(
        "一个低成本的谷歌搜索API。"
        "当你需要回答有关时事的问题时，可以调用该工具。"
        "该工具的输入是搜索查询语句。"
    ),
    args_schema=GoogleSerperArgsSchema,
    api_wrapper=GoogleSerperAPIWrapper(),
)


class GaoDeInputs(BaseModel):
    city: str = Field(description="查询天气的目标城市，例如：北京市")


class GetWeatherTool(BaseTool):
    """根据传入的天气名称查询天气"""
    name: str = "get_weather_tool"
    description: str = "使用此工具，根据城市名称查询对应天气"
    args_schema: Type[BaseModel] = GaoDeInputs

    def _get_adcode(self, city):
        df = pd.read_excel("/Users/zoe/Downloads/AMap_adcode_citycode.xlsx")  # 替换为你的文件路径
        result = df[df['中文名'] == city]['adcode']
        return result.to_string().strip().split(' ')[-1]

    def _run(self, *args: Any, **kwargs: Any) -> Any:
        gaode_api_key: str = os.getenv("GAODE_API_KEY")
        if not gaode_api_key:
            return f"高德apikey未配置"
        city: str = kwargs.get("city", "")
        ad_code = self._get_adcode(city)
        session = requests.session()
        weather_response = session.request(
            method="GET",
            url=f"https://restapi.amap.com/v3/weather/weatherInfo?key={gaode_api_key}&city={ad_code}&extensions=all",
            headers={"Content-Type": "application/json; charset=utf-8"},
        )
        weather_response.raise_for_status()
        weather_data = weather_response.json()

        if weather_data.get("info") == "OK":
            # 5. 返回最后的结果字符串
            return json.dumps(weather_data, ensure_ascii=False, indent=2)
        else:
            return f"获取 {city} 天气预报信息失败：{weather_data.get('info')}"


gd = GetWeatherTool()

# 1.工具组装为字典(工具名作为键，工具作为值)
tool_dict = {
    gd.name: gd,
    google_serper.name: google_serper,
}
# 2.组装成工具列表
tools = [tool for tool in tool_dict.values()]
# 3.模型
prompt = ChatPromptTemplate.from_messages([
    ("system", "回答用户问题，必要时调用工具"),
    ("human", "{query}")
])
llm = ChatOpenAI(model="gpt-3.5-turbo", base_url=os.getenv("OPENAI_API_BASE"))
# 4.模型绑定tools
llm_with_tools = llm.bind_tools(tools=tools)
# chain = {"query": RunnablePassthrough()} | prompt | llm
chain2 = {"query": RunnablePassthrough()} | prompt | llm_with_tools
query = "北京市今天穿什么衣服合适"
res = chain2.invoke(query)
print(res)
tool_calls = res.tool_calls
if len(tool_calls) <= 0:
    print(res.content)
else:
    # 将历史消息进行整合
    messages = prompt.invoke(query).to_messages()
    messages.append(res)
    # 调用工具
    for tool_call in tool_calls:
        tool = tool_dict.get(tool_call.get("name"))
        print("正在执行代码：", tool_call.get("name"))
        content = tool.invoke(tool_call.get("args"))
        print("工具返回结果", content)
        tool_call_id = tool_call.get("id")
        messages.append(ToolMessage(content=content, tool_call_id=tool_call_id))
    print(llm.invoke(messages).content)
```

# 2.组装成工具列表

以下是每个工具的名称和描述：

{rendered_tools}

根据用户输入，返回要使用的工具的名称和输入。
将您的响应作为具有 `name` 和 `arguments` 键的JSON块返回。
`arguments` 应该是一个字典，其中键对应于参数名称，值对应于请求的值。"""

    ("system", system_prompt),
    ("human", "{query}")

query = {"query": "北京市今天穿什么衣服合适"}
print(prompt.invoke(query))


class ToolInput(TypedDict):
    name: str
    arguments: Dict[str, Any]


def invoke_tool(
        tool_call_request: ToolCallRequest,
        config: Optional[RunnableConfig] = None,
) -> str:
    """
    我们可以使用的执行工具调用的函数。

    :param tool_call_request: 一个包含键名和参数的字典，名称必须与现有的工具名称匹配，参数是该工具的参数
    :param config: 这是LangChain中包含回调、元数据等信息的配置信息。
    :return: 工具执行的结果。
    """
    name = tool_call_request["name"]
    requested_tool = tool_dict.get(name)
    return requested_tool.invoke(tool_call_request.get("arguments"), config=config)


chain = prompt | llm | JsonOutputParser() | RunnablePassthrough.assign(output=invoke_tool)
res = chain.invoke(query)
print(res)
