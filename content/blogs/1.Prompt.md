# 1.prompt template

prompt组件中，默认使用f-string方法格式化变量。PromptTemplate是可填充变量的字符串模板。ChatPromptTemplate 输出的是“消息结构”，不是字符串。

## 1.1.PromptTemplate输入单字符串,适用文本补全模型

```python
# from_template
prompt = PromptTemplate.from_template("讲个{topic}相关的{content}")#<class 'langchain_core.prompts.prompt.PromptTemplate'>
prompt.pretty_print()
# 1.partial(输入部分变量，输出新的 Template)
prompt_partial = prompt.partial(topic="股票")#<class 'langchain_core.prompts.prompt.PromptTemplate'>
# 2.invoke(输入字典，输出PromptValue)，PromptValue 才是 LangChain 世界里的“Prompt 本体”
prompt_invoke = prompt.invoke({"topic": "股票", "content": "诗歌"})#<class 'langchain_core.prompt_values.StringPromptValue'>
print(prompt_invoke.to_string())  #讲个股票相关的诗歌
print(prompt_invoke.to_messages())  #[HumanMessage(content='讲个股票相关的诗歌', additional_kwargs={}, response_metadata={})]
for i in prompt_invoke.to_messages():
    i.pretty_print()
# 3.format(输入变量,输出str)
prompt_format = prompt.format(topic="股票", content="笑话")#<class 'str'>
```

# 2.selecotrs

ExampleSelector 类型：LengthBasedExampleSelector根据长度选几个，SemanticSimilarityExampleSelector根据向量相似度选示例，结合 FewShotPromptTemplate + SelectorLangChain 会自动选相似 example，拼接 prompt，返回最终字符串。

## 2.1.FewShotPromptTemplate

```python
examples = [
    {"input": "happy", "output": "sad"},
    {"input": "tall", "output": "short"},
    {"input": "sunny", "output": "gloomy"},
    {"input": "windy", "output": "calm"},
    {"input": "高兴", "output": "悲伤"}
]
example_prompt = PromptTemplate(
    input_variables=["input", "output"],
    template="原词: {input}\n反义: {output}"
)
dynamic_prompt = FewShotPromptTemplate(
    examples=examples,
    # example_selector=example_selector,
    example_prompt=example_prompt,
    prefix="给出每个输入词的反义词",
    suffix="原词: {adjective}\n反义: []",
    input_variables=["adjective"]
)
print(dynamic_prompt.invoke({"adjective": "big"}))
```

```python
# 动态选择实例 - 根据长度动态选择提示词示例
example_selector = LengthBasedExampleSelector(
    examples=examples,
    example_prompt=example_prompt,
    max_length=10,
)
```

```python
#动态选择实例 - 根据语义相似度选择提示词示例
dotenv.load_dotenv()
example_selector = SemanticSimilarityExampleSelector.from_examples(
    examples,
    OpenAIEmbeddings(openai_api_base=os.getenv("OPENAI_API_BASE")),
    Chroma,
    k=1,
)
```

```python
#动态选择实例 - 根据MMR与最大余弦相似度选择示例
example_selector = MaxMarginalRelevanceExampleSelector.from_examples(
    examples,
    OpenAIEmbeddings(openai_api_base=os.getenv("OPENAI_API_BASE")),
    Chroma,
    k=2,
)
```

```python
dynamic_prompt = FewShotPromptTemplate(
    example_selector=example_selector,
    example_prompt=example_prompt,
    prefix="给出每个输入词的反义词",
    suffix="原词: {adjective}\n反义: []",
    input_variables=["adjective"]
)
print(dynamic_prompt.format(adjective="big"))
```

## 2.2.FewShotChatMessagePromptTemplate

```python
examples = [
    {"input": "2 🐍 2", "output": "4"},
    {"input": "2 🐍 3", "output": "5"}
]
example_prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个厉害的人工智能助手"),
    ("human", "{input}"),
    ("ai", "{output}")
])
few_shot_prompt = FewShotChatMessagePromptTemplate(
    example_prompt=example_prompt,
    examples=examples,
)
print(few_shot_prompt.__class__)
print(few_shot_prompt.invoke({}).to_messages())
```

# 3.smith管理prompt

```python
# 使用管理[LangSmith的提示词模版](https://smith.langchain.com/hub)
# 1.创建并上传提示词到 LangSmith
# client = Client(api_key="lsv2_pt_xxxxx")  #创建api
client = Client()
prompt = ChatPromptTemplate.from_messages(
    [
        ("system", "你是一个专业的职业规划顾问。"),
        ("user", "请为拥有 {background} 背景的人提供 3 条职业建议，并说明理由。"),
    ]
)

client.push_prompt(
    "career-advice-prompt",
    object=prompt,
    description="根据用户背景给出职业建议的中文 prompt",
    tags=["career", "cn", "demo"],
)
```

```python
# 2.从 LangSmith 拉取提示词并测试
prompt = client.pull_prompt("career-advice-prompt")
formatted_prompt = prompt.invoke({"background": "agent项目"})  # 使用变量填充提示词
```

```python
# 3.更新提示词（版本迭代）
new_prompt = ChatPromptTemplate.from_messages([
    ("system", "你是一个有趣的机器人。使用中文回复。"),
    ("user", "{question}"),
])
client.push_prompt("career-advice-prompt", object=new_prompt, tags=["v2"])
```
