OutputParser的使用其实也就三步：

1.定义parser；2.parser.get_format_instructions()插入到prompt；3.输出parser.invoke()

#  1.DatetimeOutputParser

```python

# 1.创建Parser
output_parser = DatetimeOutputParser()
# 2.创建提示模板
prompt = ChatPromptTemplate.from_messages([
    ("system", f"你必须按照以下格式返回日期时间：{output_parser.get_format_instructions()}"),
    ("human", "请将以下自然语言转换为标准日期时间格式：\n###{text}###")
])
# 3.parser有两个方法解析：invoke和parse，invoke底层还是parse
resp = output_parser.invoke(chat_model.invoke(prompt.invoke({"text": "今天2026年1月22日"})))#invoke(string,message)
resp = output_parser.parse(chat_model.invoke(prompt.invoke({"text": "今天2026年1月22日"})).content)#parse(string)

```

# 2.EnumOutputParser

```python
# 0.定义枚举类
class Status(Enum):
    SUCCESS = "success"
    FAILED = "failed"
    PENDING = "pending"
# 1.创建解析器
output_parser = EnumOutputParser(enum=Status)
# 2.提示模板
prompt = ChatPromptTemplate.from_messages([
    ("system", f"判断用户现在的状态：{output_parser.get_format_instructions()}"),
    ("human", "{input}")
])
# 3.测试输入
result = chain.invoke({"input": "我现在正在失业中"})#<enum 'Status'>
```

# 3.StrOutputParser

```python
# 1.创建解析器
output_parser = StrOutputParser() 
# 2.提示模板，不用get_format_instructions()
prompt = ChatPromptTemplate.from_template("{query}")
# 3.输出invoke
chain1 = prompt | chat_model
chain2 = prompt | chat_model | output_parser
print(chain1.invoke({"query": "你是谁"}))
print(chain2.invoke({"query": "你是谁"}))
```

# 4.JsonOutputParser

```python
# 0. 创建一个json数据结构，用于告诉大语言模型这个json长什么样子
class Joke(BaseModel):
    joke: str = Field(description="回答用户的冷笑话")
    punchline: str = Field(description="这个冷笑话的笑点")
# 1.创建解析器
output_parser = JsonOutputParser(pydantic_object=Joke)
# 2.创建提示模板
prompt = ChatPromptTemplate.from_template("根据用户问题回答。\n{format_instructions}\n{query}").partial(
    format_instructions=output_parser.get_format_instructions())
# 3.输出invoke
now_prompt = prompt.invoke({"query": "给我讲个笑话"})
```

# 5.PydanticOutputParser

```python
# 0. 定义 Pydantic 模型
class User(BaseModel):
    name: str
    age: int
# 1.创建解析器
output_parser = PydanticOutputParser(pydantic_object=User)
prompt = ChatPromptTemplate.from_template(
    """
    请从用户描述中提取信息，并严格按格式输出：
    {format_instructions}

    用户描述：
    {query}
    """
)
# 2.创建提示模板&3.输出invoke
now_prompt = prompt.invoke({
    "query": "小明今年23岁",
    "format_instructions": output_parser.get_format_instructions()
})
```

# 6.通过with_structured_output绑定输出结构给模型

```python
# 0.定义结构化输出模型：QAExtra
class QAExtra(BaseModel):
    """一个问答键值对工具，传递对应的假设性问题+答案"""
    question: str = Field(description="假设性问题")
    answer: str = Field(description="假设性问题对应的答案")
# 1.通过with_structured_output绑定输出结构给模型
structured_llm = llm.with_structured_output(
    QAExtra)  # 强制输出为 QAExtra 格式,默认method == "json_schema"，内部使用PydanticOutputParser进行了转换
prompt = ChatPromptTemplate.from_messages([
    ("system", "请从用户传递的 query 中提取出假设性的问题+答案。"),
    ("human", "{query}")
])
chain = {"query": RunnablePassthrough()} | prompt | structured_llm
# 2.invoke调用带outputparser的模型
print(chain.invoke("我叫小z，我喜欢打篮球，游泳"))
```
